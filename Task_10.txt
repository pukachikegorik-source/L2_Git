function bingo(ticket, win){
    let min_wins = 0;
  for (let i = 0; i < ticket.length; i++){
    let str = ticket[i][0];
    let num = ticket[i][1];
    
    for (let j = 0; j < str.length; j++){
      if (str.charCodeAt(j) === num){
        min_wins++;
        break;
      }
    }
  }
  return min_wins >= win ? 'Winner!' : 'Loser!';
}
console.log(bingo([['ABC', 65]], 1));
console.log(bingo([['ABC', 69]], 1));
