function primeFactors(n){
  let res = '';
  let div = 2;
  
  while (n > 1){
    let count = 0;
    while (n % div === 0){
      n = n / div;
      count++;
    }
    if (count > 0){
      res += count === 1
      ? `(${div})`
      : `(${div}**${count})`;
    }
    div++;
  }
      return res;
}
    
console.log(primeFactors(12));
console.log(primeFactors(86240));
